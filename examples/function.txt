##############################################
# FUNCTION APP
##############################################

resource "azapi_resource" "loose_file_move" {
  count = var.loose_file_move_enabled ? 1 : 0

  type      = "Microsoft.Web/sites@2025-03-01"
  name      = local.loose_file_move_names.function_app
  parent_id = azurerm_resource_group.mft_rg.id
  location  = var.location

  kind = "functionapp"

  identity {
    type = "SystemAssigned, UserAssigned"
    identity_ids = [
      data.azurerm_user_assigned_identity.loose_file_move[0].id
    ]
  }

  body = {
    properties = {
      serverFarmId = azurerm_service_plan.loose_file_move[0].id

      # Secure unique default hostname.
      #
      # Example:
      # myfunc-a3dze9hpc5cucybj.eastus-01.azurewebsites.net
      autoGeneratedDomainNameLabelScope = "TenantReuse"

      httpsOnly           = true
      publicNetworkAccess = "Enabled"

      siteConfig = {
        alwaysOn           = true
        vnetRouteAllEnabled = true

        minTlsVersion = "1.2"
        ftpsState     = "Disabled"

        # AzureRM's builtin_logging_enabled = false is effectively
        # represented by not enabling the legacy App Service logging
        # configuration here.

        ipSecurityRestrictionsDefaultAction = "Deny"

        ipSecurityRestrictions = [
          {
            name      = "AllowEventGrid"
            priority  = 100
            action    = "Allow"
            tag       = "ServiceTag"
            ipAddress = "AzureEventGrid"
          }
        ]

        scmIpSecurityRestrictionsUseMain      = false
        scmIpSecurityRestrictionsDefaultAction = "Deny"

        scmIpSecurityRestrictions = [
          for idx, ip_range in var.function_scm_allowed_ip_ranges : {
            name      = "AllowDeploy${idx}"
            ipAddress = ip_range
            priority  = 100 + idx
            action    = "Allow"
          }
        ]

        # Windows Functions .NET runtime.
        #
        # See notes below regarding isolated vs in-process.
        netFrameworkVersion = var.function_dotnet_version

        appSettings = [
          {
            name  = "FUNCTIONS_EXTENSION_VERSION"
            value = "~4"
          },
          {
            name  = "WEBSITE_RUN_FROM_PACKAGE"
            value = "1"
          },

          # Host storage using SYSTEM-ASSIGNED identity.
          {
            name  = "AzureWebJobsStorage__accountName"
            value = azurerm_storage_account.loose_file_move_runtime[0].name
          },
          {
            name  = "AzureWebJobsStorage__credential"
            value = "managedidentity"
          },

          # Required for .NET isolated worker.
          {
            name  = "FUNCTIONS_WORKER_RUNTIME"
            value = var.function_use_isolated_runtime ? "dotnet-isolated" : "dotnet"
          }
        ]
      }
    }
  }

  tags = merge(local.base_tags, {})

  # We need the generated system-assigned identity principalId for RBAC.
  response_export_values = [
    "identity.principalId",
    "properties.defaultHostName"
  ]

  depends_on = [
    azurerm_private_endpoint.loose_file_move_runtime,
    azurerm_storage_account_network_rules.loose_file_move_runtime,
  ]
}
